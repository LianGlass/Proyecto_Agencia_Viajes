const
    SENSORS_PER_WAGON: integer = 14;      {Total number of sensors per wagon}
    SENSORS_PER_AXLE: integer = 7;        {Number of sensors allocated per axle}
    CRITICAL_TEMPERATURE: real = 90.0;    {Threshold temperature for alert triggering}
                                          {(Celsius)}
    VIBRATION_ALERT_LEVEL: real = 7.1;    {Threshold vibration for alert triggering}
                                          {(mm/s)}
    WEAR_CRITICAL_THRESHOLD: real = 25.0; {Minimum residual life threshold for wear}
                                          {alert (%)}
    MIN_WORN_FOR_WARNING: integer = 3;    {Minimum worn sensors count to declare wagon}
                                          {unreliable}
    COMPONENT_TYPES: integer = 4;         {Total number of distinct component types}
end const

type
    tComponentType = {BEARING, WHEEL, BRAKE, AXLE};

    {Record for an individual sensor data}
    tSensor = record
        componentType: tComponentType; {Mechanical component monitored by the sensor}
        temperature: real;             {Temperature reading in Celsius}
        vibration: real;               {Vibration level in mm/s}
        residualLife: real;            {Remaining component life percentage}
    end record

    {Array of tuples for storing wagon sensors}
    tWagonSensors = vector[SENSORS_PER_WAGON] of tSensor;

    {Array for alert counters per component type}
    tAlertCounts = vector[COMPONENT_TYPES] of integer;

    {Tuple grouping inspection results}
    tInspectionResult = record
        avgTemp: real;             {Average temperature across all wagon sensors}
        maxTemp: real;             {Highest temperature recorded in the inspection}
        peakPos: integer;          {Array index of the sensor with maximum temperature}
        alertCount: integer;       {Total number of sensors in alert status}
        wornCount: integer;        {Total number of worn-out sensors}
        alertCounts: tAlertCounts; {Count of alerts categorized by component type}
    end record
end type

{--- HEADER / DECLARATIONS OF FUNCTIONS AND ACTIONS ---}

{Exercise 1}
{Loads sensor data from file into the tuple vector}
action loadWagonFromFile(in filename: string, out sensors: tWagonSensors,
    out isRead:boolean)
    var
        fichero: file;     {fichero que se abre}
        i: integer;        {variable de iteracion}
    end var

    fichero := openFile(filename);

    {si el fichero no se ha podido abrir no se lee nada}
    if fichero = null then
        isRead := false;
    else
        {lectura de los catorce sensores}
        i := 1;
        while i <= SENSORS_PER_WAGON do
            sensors[i].componentType := readEnumFromFile(fichero);
            sensors[i].temperature := readRealFromFile(fichero);
            sensors[i].vibration := readRealFromFile(fichero);
            sensors[i].residualLife := readRealFromFile(fichero);
            i := i + 1;
        end while
        closeFile(fichero);
        isRead := true;
    end if
end action

{Exercise 2}
{Processes wagon inspection and packages the output into result tuple}
action inspectWagon(in sensors: tWagonSensors, out result: tInspectionResult)
    var
        i: integer;            {variable de iteracion}
        indice: integer;       {indice del tipo de componente en alerta}
        sumaTemp: real;        {suma de las temperaturas}
    end var

    {inicializacion de acumuladores y contadores}
    sumaTemp := 0.0;
    result.maxTemp := sensors[1].temperature;
    result.peakPos := 1;
    result.alertCount := 0;
    result.wornCount := 0;
    i := 1;
    while i <= COMPONENT_TYPES do
        result.alertCounts[i] := 0;
        i := i + 1;
    end while

    i := 1;
    while i <= SENSORS_PER_WAGON do
        {acumula la temperatura}
        sumaTemp := sumaTemp + sensors[i].temperature;

        {temperatura mas alta: con > se queda la primera posicion si se repite}
        if sensors[i].temperature > result.maxTemp then
            result.maxTemp := sensors[i].temperature;
            result.peakPos := i;
        end if

        {cuenta los desgastados}
        if isSensorWorn(sensors[i]) then
            result.wornCount := result.wornCount + 1;
        end if

        {cuenta las alertas, en total y por tipo de componente}
        if isSensorInAlert(sensors[i]) then
            result.alertCount := result.alertCount + 1;
            indice := getComponentTypeIndex(sensors[i].componentType);
            result.alertCounts[indice] := result.alertCounts[indice] + 1;
        end if

        i := i + 1;
    end while

    {temperatura media}
    result.avgTemp := sumaTemp / SENSORS_PER_WAGON;
end action

{Exercise 2.1}
{Returns true if the sensor residual life is critical}
function isSensorWorn(sensor: tSensor): boolean
    var
        desgastado: boolean;   {resultado de la comprobacion}
    end var

    desgastado := sensor.residualLife < WEAR_CRITICAL_THRESHOLD;

    return desgastado;
end function

{Exercise 2.2}
{Returns true if the sensor exceeds temperature/vibration or is worn}
function isSensorInAlert(sensor: tSensor): boolean
    var
        enAlerta: boolean;     {resultado de la comprobacion}
    end var

    enAlerta := sensor.temperature > CRITICAL_TEMPERATURE
                or sensor.vibration > VIBRATION_ALERT_LEVEL
                or isSensorWorn(sensor);

    return enAlerta;
end function

{Exercise 2.3}
{Returns the vector index corresponding to a component type}
function getComponentTypeIndex(compType: tComponentType): integer
    var
        indice: integer;       {indice que se devuelve}
    end var

    {los indices 1 a 4 los fija el enunciado}
    if compType = BEARING then
        indice := 1;
    else if compType = WHEEL then
        indice := 2;
    else if compType = BRAKE then
        indice := 3;
    else
        indice := 4;
    end if

    return indice;
end function

{Exercise 3}
{Displays inspection summary on screen}
action displaySummary(in sensors: tWagonSensors, in result: tInspectionResult)
    var
        i: integer;            {variable de iteracion}
    end var

    writeString("----------------------------------------");
    writeString("INSPECTION SUMMARY");
    writeString("----------------------------------------");
    writeString("COMPONENT TYPES: 1-BEARING, 2-WHEEL, 3-BRAKE, 4-AXLE");
    writeString("AVERAGE TEMPERATURE (CELSIUS): ");
    writeReal(result.avgTemp);
    writeString("PEAK TEMPERATURE (CELSIUS): ");
    writeReal(result.maxTemp);
    writeString(" AT POSITION ");
    writeInteger(result.peakPos);
    writeString(" (AXLE: ");
    writeString(getSensorAxleName(result.peakPos));
    writeString(", TYPE: ");
    writeString(getComponentName(sensors[result.peakPos].componentType));
    writeString(")");

    {componentes en alerta}
    if result.alertCount = 0 then
        writeString("NO COMPONENTS IN ALERT");
    else
        writeString("COMPONENTS IN ALERT: ");
        writeInteger(result.alertCount);

        {lista en orden de posicion: se muestran los sensores que estan en alerta}
        i := 1;
        while i <= SENSORS_PER_WAGON do
            if isSensorInAlert(sensors[i]) then
                writeString("POSITION ");
                writeInteger(i);
                writeString(" - TYPE: ");
                writeString(getComponentName(sensors[i].componentType));
            end if
            i := i + 1;
        end while

        {alertas por tipo}
        writeString("ALERTS BY COMPONENT TYPE:");
        writeString("- BEARING: ");
        writeInteger(result.alertCounts[getComponentTypeIndex(BEARING)]);
        writeString("- WHEEL: ");
        writeInteger(result.alertCounts[getComponentTypeIndex(WHEEL)]);
        writeString("- BRAKE: ");
        writeInteger(result.alertCounts[getComponentTypeIndex(BRAKE)]);
        writeString("- AXLE: ");
        writeInteger(result.alertCounts[getComponentTypeIndex(AXLE)]);
    end if

    {desgastados y estado del vagon}
    writeString("WORN COMPONENTS: ");
    writeInteger(result.wornCount);
    if result.wornCount >= MIN_WORN_FOR_WARNING then
        writeString("WAGON STATUS: NOT RELIABLE");
    else
        writeString("WAGON STATUS: RELIABLE");
    end if
    writeString("----------------------------------------");
end action

{Exercise 3.1}
{Returns the axle identifier string based on sensor position}
function getSensorAxleName(position: integer): string
    var
        nombre: string;        {nombre que se devuelve}
    end var

    if position <= SENSORS_PER_AXLE then
        nombre := "AXLE1";
    else
        nombre := "AXLE2";
    end if

    return nombre;
end function

{Exercise 3.2}
{Returns the string representation of a component type}
function getComponentName(compType: tComponentType): string
    var
        nombre: string;        {nombre que se devuelve}
    end var

    if compType = BEARING then
        nombre := "BEARING";
    else if compType = WHEEL then
        nombre := "WHEEL";
    else if compType = BRAKE then
        nombre := "BRAKE";
    else
        nombre := "AXLE";
    end if

    return nombre;
end function

algorithm WagonInspectionModular
    var
        sensors: tWagonSensors;               {Array storing all sensor data of the wagon}
        filename: string;                     {Name of the input data file}
        inspectionResult: tInspectionResult;  {Tuple containing all calculated}
                                              {inspection metrics}
        isRead: boolean;                      {indica si la lectura del fichero fue correcta}
    end var

    writeString("--- WAGON INSPECTION SYSTEM (PR2 - MODULAR) ---");
    writeString("LOAD DATA FROM FILE. ENTER FILE NAME >> ");
    filename := readString();

    {Modular action calls}
    {Exercise 1}
    loadWagonFromFile(filename, sensors, isRead);

    {Exercise 2}
    if isRead then
        inspectWagon(sensors, inspectionResult);
    end if

    {Exercise 3}
    if isRead then
        displaySummary(sensors, inspectionResult);
    else
        writeString("Error: Could not open file");
    end if
end algorithm
